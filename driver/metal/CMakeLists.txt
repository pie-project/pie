# This driver is Metal 4 only: MTL4* command queues, argument tables, explicit
# barriers, residency sets and placement-sparse heaps have no pre-macOS-26
# equivalent here, and the kernels are compiled as MSL 4.0. Pinning the
# deployment target makes that a build-time contract rather than a runtime
# surprise, and lets the @available-free source compile without warnings.
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING
      "Minimum macOS version (Metal 4 floor)" FORCE)
endif()

# `-Werror=switch` on forward.cpp alone.
#
# `MetalExecutor::setup` switches on the model family with no `default:`, so
# adding a family without teaching the executor about it must not compile. It
# was a plain warning first, and a warning is exactly what a gpt-oss checkpoint
# fell through: it matched no case, ran on past the switch, and was set up as
# qwen3.5 -- surfacing as "unstaged weight shared_embedding.weight", a tensor
# belonging to a different family entirely.
#
# Scoped to this file rather than the target: the switches over the shared
# `Kernel` enum are legitimately partial (a family handles its own kinds and
# falls through for the rest), and making those an error would only buy a
# forest of `default:` labels that turn the check back off.
set_source_files_properties(src/batch/simple_family.cpp
  PROPERTIES COMPILE_OPTIONS "-Werror=switch")
set_source_files_properties(src/batch/forward.cpp
  PROPERTIES COMPILE_OPTIONS "-Werror=switch")

# Compiled-in fallback for the native .metal kernel sources (runtime PSO
# compilation). Overridable at run time via PIE_METAL_KERNELS_DIR. PUBLIC so
# every target linking pie_driver_metal_lib (e.g. ptir_checkpoint_e2e_test,
# which drives an independent MetalExecutor::setup outside Context) sees
# the same default when PIE_METAL_KERNELS_DIR is unset. Installed packaging
# of the kernels dir is a follow-up.
#
# ptir_rng.generated.metal is owned by pie-codegen and staged here at configure
# time rather than checked in twice. The host emitter now embeds the M1 runtime
# template into every emitted MSL source itself, but the RNG preamble still
# needs to sit beside the checked-in .metal kernels so a build-time
# `configure_file` (below) keeps them in lockstep -- the driver splices it in
# at compile time because Metal's runtime shader compiler does no filesystem
# include lookup.
configure_file(
  ${CMAKE_CURRENT_SOURCE_DIR}/../../compiler/codegen/include/ptir_rng.generated.metal
  ${CMAKE_CURRENT_SOURCE_DIR}/src/kernels/ptir_rng.generated.metal
  COPYONLY)

# ---- the loader archive ----------------------------------------------------
#
# `src/batch/forward.cpp` compiles a LoadPlan, which means calling into the
# loader's C ABI — `pie-loader-capi`, an rlib the *worker* links, not something
# a C++ binary can pull in. `cargo rustc -p pie-loader-capi --lib --crate-type
# staticlib` emits an archive that does link. Anything that links
# `pie_driver_metal_lib` into an executable needs it; the archive target itself
# does not, which is why `worker/build.rs` (it builds only
# `pie_driver_metal_lib`) never noticed.
#
# The name is `pie_loader_capi` and NOT `pie_loader`: the C entry points moved
# out of the compiler crate when authoring did, so a `-p pie-loader` staticlib
# still builds and still links and exports none of the eighteen `pie_loader_*`
# symbols. Searching for the old name found that archive and failed at link
# with an undefined `_pie_loader_verify_model` instead of skipping.
#
# Mirrors driver/cuda: when the archive is absent those targets are skipped
# rather than failing the build for everyone who has not run that command.
if(NOT PIE_LOADER_STATICLIB)
  find_library(PIE_LOADER_STATICLIB
    NAMES pie_loader_capi
    PATHS ${CMAKE_CURRENT_SOURCE_DIR}/../../target/debug
          ${CMAKE_CURRENT_SOURCE_DIR}/../../target/release
    NO_DEFAULT_PATH)
endif()

function(add_metal_host_test target source)
  add_executable(${target} ${source})
  # `${PIE_DRIVER_ABI_INCLUDE_DIR}` carries the generated `pie_driver_abi.h`,
  # which `interp.hpp` and `launch/program.hpp` both include. Every other target
  # here lists it; without it these two do not compile.
  target_include_directories(${target} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/../common/include ${CMAKE_CURRENT_SOURCE_DIR}/../../compiler/codegen/include
    ${PIE_DRIVER_ABI_INCLUDE_DIR})
  add_test(NAME ${target} COMMAND ${target})
endfunction()

# Tests that link the archive. Whether a given one drags in `forward.cpp`'s
# loader calls is a link-time detail that moves, so the archive is offered to
# all of them when it exists; `NEEDS_LOADER` marks the ones that provably need
# it, and those are skipped when it does not — the same trade driver/cuda makes.
function(add_metal_linked_test target source)
  cmake_parse_arguments(ARG "NEEDS_LOADER" "" "" ${ARGN})
  if(ARG_NEEDS_LOADER AND NOT PIE_LOADER_STATICLIB)
    message(STATUS
      "${target}: skipped (no libpie_loader_capi.a; build one with "
      "`cargo rustc -p pie-loader-capi --lib --crate-type staticlib`)")
    return()
  endif()
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE pie_driver_metal_lib)
  if(PIE_LOADER_STATICLIB)
    target_link_libraries(${target} PRIVATE ${PIE_LOADER_STATICLIB})
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()

  # A stopwatch that checks its own work. Under ctest it SKIPs (no checkpoint
  # path), so the suite stays green on a machine without an 1.1 GB download;
  # given one it asserts the greedy continuation against mlx-lm's and prints
  # throughput. The timings are printed, never asserted -- a number that depends

  add_metal_linked_test(llama_bench tests/llama_bench.cpp NEEDS_LOADER)
  if(TARGET llama_bench)
    target_compile_definitions(llama_bench PRIVATE
      PIE_METAL_KERNELS_DIR_FOR_TEST="${CMAKE_CURRENT_SOURCE_DIR}/src/kernels")
  endif()
  add_metal_linked_test(simple_family_executor_test tests/simple_family_executor_test.cpp NEEDS_LOADER)
  if(TARGET simple_family_executor_test)
    target_compile_definitions(simple_family_executor_test PRIVATE
      PIE_METAL_KERNELS_DIR_FOR_TEST="${CMAKE_CURRENT_SOURCE_DIR}/src/kernels")
  endif()
  # The same cases again with the weights STREAMED rather than copied. Streaming
  # is meant to change what is resident and nothing else, so the assertions are
  # the same assertions -- if it ever changes an answer, the test that says so
  # is the one that already knows every answer.
  if(TARGET simple_family_executor_test)
    add_test(NAME simple_family_executor_streamed_test
             COMMAND simple_family_executor_test)
    set_tests_properties(simple_family_executor_streamed_test PROPERTIES
      ENVIRONMENT "PIE_METAL_TEST_STREAM_EXPERTS=1")
  endif()
  add_metal_linked_test(gemma4_prefill_numerics_test tests/gemma4_prefill_numerics_test.cpp NEEDS_LOADER)
  if(TARGET gemma4_prefill_numerics_test)
    target_compile_definitions(gemma4_prefill_numerics_test PRIVATE
      PIE_METAL_KERNELS_DIR_FOR_TEST="${CMAKE_CURRENT_SOURCE_DIR}/src/kernels")
  endif()

  add_metal_linked_test(decode_step_mb_test tests/decode_step_mb_test.cpp)

  # `router_topk` past one simdgroup. gpt-oss's 32 experts never exercised the

  add_metal_linked_test(loader_fill_test tests/loader_fill_test.cpp)

  set(PIE_METAL_CHECKPOINT_TESTS
    gemma4_forward_test

    gptoss_forward_test

    endif()
  endforeach()
